JavaScript ES6+ Features Every Developer Should Know — Complete Beginner's Guide

July 17, 2026 11 min read

If you learned JavaScript before 2015, you remember var, string concatenation with +, and writing function declarations everywhere. Modern JavaScript (ES6 and beyond) is a dramatically different language — more expressive, more readable, and more powerful. This guide covers every ES6+ feature you'll actually use in production, with working examples you can try in your browser console right now.

1. let and const (Goodbye, var)

ES6 replaced var with let and const. Both are block-scoped (limited to the nearest { } block), unlike var which is function-scoped and can cause subtle bugs.

// ❌ var — leaks out of loops
for (var i = 0; i < 3; i++) {}
console.log(i); // 3 — i leaked out!

// ✅ let — block-scoped
for (let j = 0; j < 3; j++) {}
console.log(j); // ReferenceError: j is not defined

// ✅ const — cannot be reassigned
const name = "Hugo";
name = "Other"; // TypeError!

// const arrays/objects are still mutable
const users = ["Alice", "Bob"];
users.push("Charlie"); // ✅ Works — the array reference didn't change
console.log(users); // ["Alice", "Bob", "Charlie"]

Rule of thumb: Use const by default. Only use let when you need to reassign. Never use var.

2. Template Literals

Forget string concatenation. Template literals use backticks (`) and let you embed variables and expressions directly in strings:

// ❌ Old way
const greeting = "Hello, " + name + "! You have " + count + " messages.";

// ✅ Template literals
const greeting = `Hello, ${name}! You have ${count} messages.`;

// Expressions work too
const total = `Price: $${price * quantity} (tax: $${(price * quantity * 0.08).toFixed(2)})`;

// Multi-line strings (no more \n!)
const html = `
  <div class="card">
    <h2>${title}</h2>
    <p>${description}</p>
  </div>
`;

// Tagged templates (advanced)
function highlight(strings, ...values) {
  return strings.reduce((result, str, i) => {
    return result + str + (values[i] ? `${values[i]}` : "");
  }, "");
}

const message = highlight`Found ${count} matches in ${fileName}.`;
// <span>Found <mark>5</mark> matches in <mark>report.txt</mark>.</span>

3. Arrow Functions

Arrow functions (=>) are shorter syntax for functions that don't need their own this binding:

// ❌ Old way
const square = function(x) {
  return x * x;
};

// ✅ Arrow function (single expression auto-returns)
const square = (x) => x * x;

// Multiple parameters need parentheses
const add = (a, b) => a + b;

// Multiple lines need curly braces and explicit return
const processUser = (user) => {
  const name = user.name.trim().toLowerCase();
  const age = parseInt(user.age);
  return { name, age, valid: age >= 18 };
};

// Single parameter — no parentheses needed
const double = x => x * 2;

// Object literal return needs parentheses
const createUser = (name) => ({ name, id: Date.now() });

// Arrow functions inherit `this` from parent scope
const obj = {
  name: "Hugo",
  // Regular function — `this` is the element, not obj
  greetOld() {
    setTimeout(function() {
      console.log(this.name); // undefined
    }, 100);
  },
  // Arrow function — `this` is captured from obj
  greetNew() {
    setTimeout(() => {
      console.log(this.name); // "Hugo" ✅
    }, 100);
  }
};

4. Destructuring

Extract values from arrays and objects in one clean line:

// === Object Destructuring ===
const user = {
  name: "Hugo",
  age: 28,
  email: "hugo@example.com",
  address: { city: "San Francisco", zip: "94102" }
};

// Extract specific properties
const { name, email } = user;
console.log(name); // "Hugo"

// Rename variables
const { name: userName, age: userAge } = user;

// Default values
const { phone = "N/A" } = user; // "N/A"

// Nested destructuring
const { address: { city, zip } } = user;
console.log(city); // "San Francisco"

// Destructure remaining properties with rest
const { name, ...rest } = user;
console.log(rest); // { age: 28, email: "...", address: {...} }

// === Array Destructuring ===
const colors = ["red", "green", "blue", "yellow"];
const [first, second, , fourth] = colors;
// first = "red", second = "green", fourth = "yellow"

// Swap variables (no temp variable!)
let a = 1, b = 2;
[a, b] = [b, a];
// a = 2, b = 1

// Default values
const [one, two, three = 3, four = 4] = [1, 2];
// three = 3, four = 4

// Rest pattern
const [head, ...tail] = colors;
// head = "red", tail = ["green", "blue", "yellow"]

// === Function Parameter Destructuring ===
function greet({ name, greeting = "Hello" }) {
  return `${greeting}, ${name}!`;
}

greet({ name: "Hugo", greeting: "Hi" }); // "Hi, Hugo!"
greet({ name: "Hugo" }); // "Hello, Hugo!"

// Combine everything
function createUser({ name, email }, [role, permissions]) {
  return { name, email, role, permissions };
}

const user = createUser(
  { name: "Hugo", email: "hugo@example.com" },
  ["admin", ["read", "write", "delete"]]
);
// { name: "Hugo", email: "...", role: "admin", permissions: [...] }

5. Spread and Rest Operators

The spread operator (...) "spreads" elements into individual items. The rest operator collects remaining items into an array.

// === Spread ===

// Spread an array
const arr1 = [1, 2, 3];
const arr2 = [...arr1, 4, 5]; // [1, 2, 3, 4, 5]

// Merge arrays
const merged = [...arr1, ...arr2]; // [1, 2, 3, 1, 2, 3, 4, 5]

// Deep copy (shallow — objects inside still share references)
const original = { a: 1, b: { c: 2 } };
const copy = { ...original };
copy.a = 10;
console.log(original.a); // 1 (unchanged)
copy.b.c = 20;
console.log(original.b.c); // 20 (shared reference!)

// Spread in function calls
const nums = [1, 2, 3, 4, 5];
Math.max(...nums); // 5
Math.min(...nums); // 1

// Add class to an element
element.classList.add(...newClasses);

// === Rest ===

// Rest in function parameters
function sum(...numbers) {
  return numbers.reduce((total, n) => total + n, 0);
}
sum(1, 2, 3, 4); // 10

// Rest with destructuring
function logUser({ name, ...rest }) {
  console.log(`${name} has extra data:`, rest);
}

const [first, ...others] = [1, 2, 3, 4];
// first = 1, others = [2, 3, 4]

6. Default Parameters

Set default values for function parameters directly in the function signature:

// ❌ Old way
function createConnection(host, port, timeout) {
  host = host || "localhost";
  port = port || 3000;
  timeout = timeout || 5000;
}
// Problem: host = host || "localhost" fails if host is 0 or ""

// ✅ Default parameters
function createConnection(
  host = "localhost",
  port = 3000,
  timeout = 5000
) {
  console.log(`Connecting to ${host}:${port} with ${timeout}ms timeout`);
}

createConnection(); // localhost:3000, 5000ms
createConnection("api.example.com"); // api.example.com:3000, 5000ms
createConnection("api.example.com", 8080, 10000);

// Default values can be expressions
function greet(name, greeting = `Hello, ${name}!`) {
  return greeting;
}

7. Enhanced Object Literals

Write object literals more concisely:

// Shorthand property names (name matches variable name)
const name = "Hugo";
const age = 28;

// ❌ Old way
const user = { name: name, age: age };

// ✅ Shorthand
const user = { name, age };

// Shorthand method definitions
const calculator = {
  base: 10,
  // ❌ Old: multiply: function(x) { return this.base * x; }
  // ✅ Shorthand
  multiply(x) { return this.base * x; },
  divide(x) { return this.base / x; }
};

// Computed property names
const key = "email";
const user = {
  name: "Hugo",
  [key]: "hugo@example.com" // property name is "email"
};
console.log(user.email); // "hugo@example.com"

8. Array Methods: map, filter, reduce

Three methods that replace most for loops in JavaScript:

const users = [
  { name: "Alice", age: 25, role: "admin" },
  { name: "Bob", age: 17, role: "user" },
  { name: "Charlie", age: 30, role: "admin" },
  { name: "Diana", age: 22, role: "user" }
];

// .map() — transform each element (returns new array, same length)
const names = users.map(user => user.name);
// ["Alice", "Bob", "Charlie", "Diana"]

const upperNames = users.map(u => u.name.toUpperCase());
// ["ALICE", "BOB", "CHARLIE", "DIANA"]

// .filter() — keep elements that match a condition
const adults = users.filter(user => user.age >= 18);
// [{ name: "Alice", ... }, { name: "Charlie", ... }, { name: "Diana", ... }]

const admins = users.filter(user => user.role === "admin");
// [{ name: "Alice", ... }, { name: "Charlie", ... }]

// .reduce() — combine all elements into a single value
const totalAge = users.reduce((sum, user) => sum + user.age, 0);
// 94

const nameCount = users.reduce((acc, user) => {
  acc[user.name[0]]++;
  return acc;
}, {});
// { A: 2, B: 1, D: 1 }

// Chain them (most powerful pattern!)
const adminNames = users
  .filter(u => u.role === "admin")  // filter first
  .map(u => u.name)                // then transform
  .sort()                          // then sort
  .join(", ");                     // then join to string
// "Alice, Charlie"

9. Promises and async/await

Modern JavaScript handles asynchronous code with promises and the cleaner async/await syntax:

// === Promises ===

// Creating a promise
function fetchData(url) {
  return new Promise((resolve, reject) => {
    fetch(url)
      .then(response => {
        if (!response.ok) {
          reject(new Error(`HTTP ${response.status}`));
        }
        return response.json();
      })
      .then(data => resolve(data))
      .catch(error => reject(error));
  });
}

// Using promises with .then()
fetchData("/api/users")
  .then(users => console.log(users))
  .catch(error => console.error(error));

// === async/await (cleaner!) ===

async function loadUsers() {
  try {
    const users = await fetchData("/api/users");
    const filtered = users.filter(u => u.active);
    console.log(filtered);
  } catch (error) {
    console.error("Failed to load users:", error.message);
  }
}

// Parallel execution with Promise.all
async function loadDashboard() {
  const [users, posts, settings] = await Promise.all([
    fetchData("/api/users"),
    fetchData("/api/posts"),
    fetchData("/api/settings")
  ]);
  // All three requests run simultaneously, not sequentially!
  return { users, posts, settings };
}

// Sequential execution (wait for each before the next)
async function processUser(userId) {
  const user = await fetchData(`/api/users/${userId}`);
  const profile = await fetchData(`/api/profiles/${user.profileId}`);
  const stats = await fetchData(`/api/stats/${userId}`);
  return { ...user, profile, stats };
}

// Promise.race — return whichever finishes first
const result = await Promise.race([
  fetchData("/api/fast-endpoint"),
  new Promise((_, reject) =>
    setTimeout(() => reject(new Error("Timeout")), 5000)
  )
]);

10. ES Modules (import/export)

Organize code into modules that can be imported:

// === utilities.js (export) ===

// Named exports
export function slugify(text) {
  return text.toLowerCase().replace(/\s+/g, '-');
}

export const API_URL = "https://api.example.com";
export const VERSION = "1.0.0";

// Default export (one per file)
export default function formatDate(date) {
  return new Date(date).toLocaleDateString('en-US', {
    year: 'numeric', month: 'long', day: 'numeric'
  });
}

// === app.js (import) ===

// Import default
import formatDate from './utilities.js';

// Import named exports
import { slugify, API_URL } from './utilities.js';

// Import with renaming
import { API_URL as baseUrl } from './utilities.js';

// Import everything as namespace
import * as utils from './utilities.js';

// Usage
console.log(formatDate("2026-01-15")); // "January 15, 2026"
console.log(slugify("Hello World"));    // "hello-world"
console.log(API_URL);                  // "https://api.example.com"
console.log(utils.VERSION);            // "1.0.0"

HTML setup: To use ES modules in the browser, add type="module" to your script tag:

<!-- This enables import/export -->
<script type="module" src="app.js"></script>

<!-- Old style (no import/export support) -->
<script src="app.js"></script>

11. Optional Chaining and Nullish Coalescing

Two features that prevent the most common JavaScript errors:

const response = {
  data: {
    user: {
      profile: {
        avatar: null
      }
    }
  }
};

// === Optional Chaining (?.) ===
// ❌ Old way — crashes if any intermediate is null/undefined
// const avatar = response.data.user.profile.avatar.url;

// ✅ Safe access — returns undefined instead of crashing
const avatar = response.data?.user?.profile?.avatar?.url;
console.log(avatar); // undefined (not an error!)

// Safe function call
const name = response.data?.user?.getName?.();

// Safe array access
const first = items?.[0]?.name;

// === Nullish Coalescing (??) ===
// Returns right-hand value only if left is null or undefined (NOT falsy)
const count = response.data?.user?.count ?? 0;

// Difference from ||:
const value1 = "" ?? "default";   // "" (empty string is valid!)
const value2 = "" || "default";   // "default" (empty string is falsy!)
const value3 = 0 ?? "default";    // 0 (zero is valid!)
const value4 = 0 || "default";    // "default" (zero is falsy!)

// Combined pattern — the most common use
const avatar = response.data?.user?.profile?.avatar?.url ?? '/default-avatar.png';

12. Set and Map

Two data structures missing from pre-ES6 JavaScript:

// === Set (unique values) ===
const tags = new Set(["javascript", "python", "javascript"]);
tags.add("docker");
tags.delete("python");

console.log(tags.has("javascript")); // true
console.log(tags.size);              // 2

// Remove duplicates from an array
const unique = [...new Set([1, 2, 2, 3, 3, 4])];
// [1, 2, 3, 4]

// Set operations
const setA = new Set([1, 2, 3, 4]);
const setB = new Set([3, 4, 5, 6]);

// Intersection
const intersection = new Set([...setA].filter(x => setB.has(x)));
// {3, 4}

// Union
const union = new Set([...setA, ...setB]);
// {1, 2, 3, 4, 5, 6}

// === Map (key-value with any type keys) ===
const cache = new Map();
cache.set("users", [/* ... */]);
cache.set("posts", [/* ... */]);
cache.set(Symbol("key"), "value"); // Symbol keys work!

console.log(cache.get("users"));
console.log(cache.has("posts"));
cache.delete("posts");
console.log(cache.size); // 2

// vs Object — Map maintains insertion order, allows any key type
const obj = {};
obj["1"] = "string";
obj[1] = "number";
console.log(obj); // { "1": "number" } — keys merged!

const map = new Map();
map.set(1, "number");
map.set("1", "string");
console.log(map.size); // 2 — different keys!

13. for...of vs for...in

Two loops that do different things:

const arr = ["a", "b", "c"];
const obj = { a: 1, b: 2, c: 3 };

// for...of — iterates over VALUES
for (const value of arr) {
  console.log(value); // "a", "b", "c"
}

for (const [key, value] of Object.entries(obj)) {
  console.log(`${key}: ${value}`); // "a: 1", "b: 2", "c: 3"
}

// for...in — iterates over KEYS (use for objects)
for (const key in obj) {
  console.log(`${key}: ${obj[key]}`); // "a: 1", "b: 2", "c: 3"
}

// ❌ Don't use for...in on arrays (iterates over indices as strings)
for (const index in arr) {
  console.log(index); // "0", "1", "2" (strings, not numbers!)
}

Browser Support

All features covered in this guide are supported in all modern browsers (Chrome, Firefox, Safari, Edge) and Node.js 14+. For older browser support, use Babel or ESBuild to transpile:

# Install a transpiler
npm install -D esbuild

# Transpile ES modules to browser-compatible code
npx esbuild app.js --bundle --outfile=bundle.js

# Or for development with hot reload
npx esbuild app.js --bundle --outfile=bundle.js --watch

Master these ES6+ features and your JavaScript code will be cleaner, shorter, and less error-prone. Start by replacing var with const, using template literals instead of + concatenation, and reaching for map/filter/reduce instead of for loops. The rest will follow naturally.
