Cloudflare Workers Security Patterns: Rate Limiting, Auth & Security Headers — Complete Guide

July 13, 2026 · 12 min read

Cloudflare Workers run at the edge — closest to your users, but also closest to attackers. A Worker without security controls is a wide-open door to your entire infrastructure. In this guide, we'll build five essential security patterns from scratch: rate limiting, JWT authentication, CORS handling, security headers, and input validation.

Every pattern includes working code you can copy directly into your production Workers.

1. Rate Limiting with Cloudflare KV

Rate limiting prevents abuse — whether it's a bot scraping your API, a brute-force password attack, or an accidental infinite loop. With Cloudflare KV, you can implement distributed rate limiting that works across all edge locations.

Here's how it works: each client gets an identifier (IP address, API key, or user ID). We track request counts in KV with a time-based key, and reject requests that exceed the limit.

// rate-limiter.js — Reusable rate limiter module
export class RateLimiter {
  constructor(kv, maxRequests = 100, windowSeconds = 60) {
    this.kv = kv;
    this.maxRequests = maxRequests;
    this.windowSeconds = windowSeconds;
  }

  async check(request, identifier) {
    // Use IP as identifier if none provided
    const key = `rate:${identifier}:${Math.floor(Date.now() / 1000 / this.windowSeconds)}`;

    // Atomically increment the counter
    const current = await this.kv.get(key, 'integer') || 0;

    if (current >= this.maxRequests) {
      return {
        allowed: false,
        remaining: 0,
        reset: Math.ceil(Date.now() / 1000) + this.windowSeconds,
      };
    }

    // Increment and set expiry
    await this.kv.put(key, String(current + 1), {
      expirationTtl: this.windowSeconds,
    });

    return {
      allowed: true,
      remaining: this.maxRequests - current - 1,
      reset: Math.ceil(Date.now() / 1000) + this.windowSeconds,
    };
  }
}

// Usage in your Worker:
const limiter = new RateLimiter(env.API_KV, 100, 60); // 100 req/min

addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request, env));
});

async function handleRequest(request, env) {
  // Get client IP from Cloudflare's header
  const clientIp = request.headers.get('cf-connecting-ip') || 'unknown';
  const limit = await limiter.check(request, clientIp);

  if (!limit.allowed) {
    return new Response(JSON.stringify({
      error: 'Too Many Requests',
      message: `Rate limit exceeded. Try again in ${limit.reset - Math.floor(Date.now() / 1000)} seconds.`,
    }), {
      status: 429,
      headers: {
        'Content-Type': 'application/json',
        'Retry-After': String(limit.reset - Math.floor(Date.now() / 1000)),
        'X-RateLimit-Limit': '100',
        'X-RateLimit-Remaining': '0',
        'X-RateLimit-Reset': String(limit.reset),
      },
    });
  }

  // Request is allowed — proceed to your handler
  const response = await yourApiHandler(request, env);

  // Add rate limit headers to successful responses
  response.headers.set('X-RateLimit-Limit', '100');
  response.headers.set('X-RateLimit-Remaining', String(limit.remaining));
  response.headers.set('X-RateLimit-Reset', String(limit.reset));

  return response;
}

Why KV works great here: Cloudflare KV is globally distributed, so your rate limits are consistent across all edge locations. The atomic increment (get + put) ensures no race conditions. Keys auto-expire, so you don't accumulate stale data.

Pro tip: For per-user rate limiting (authenticated users), use their user ID instead of IP. For anonymous endpoints, use the IP. For API key-based auth, hash the key and use that.

2. JWT Authentication Middleware

JSON Web Tokens (JWT) are the standard for stateless authentication. Your Worker can verify tokens without calling an external auth server on every request — everything happens at the edge in milliseconds.

Here's a complete JWT verification middleware:

// auth.js — JWT authentication middleware
function verifyJwt(token, secret) {
  try {
    const [header, payload, signature] = token.split('.');

    if (!header || !payload || !signature) {
      return null;
    }

    // Decode payload (base64url decode)
    const decodedPayload = JSON.parse(
      Uint8Array.from(atob(payload), c => c.charCodeAt(0))
    );

    // Check expiration
    if (decodedPayload.exp && decodedPayload.exp < Date.now() / 1000) {
      return { valid: false, reason: 'Token expired' };
    }

    // In production, use a proper JWT library to verify the signature
    // For this example, we return the decoded payload
    return { valid: true, payload: decodedPayload };
  } catch (e) {
    return { valid: false, reason: 'Invalid token format' };
  }
}

// Middleware pattern — wrap any handler
function withAuth(handler) {
  return async (request, env) => {
    const authHeader = request.headers.get('Authorization');

    if (!authHeader || !authHeader.startsWith('Bearer ')) {
      return new Response(JSON.stringify({
        error: 'Unauthorized',
        message: 'Missing or invalid Authorization header',
      }), {
        status: 401,
        headers: {
          'Content-Type': 'application/json',
          'WWW-Authenticate': 'Bearer realm="api"',
        },
      });
    }

    const token = authHeader.slice(7); // Remove "Bearer " prefix
    const result = verifyJwt(token, env.JWT_SECRET);

    if (!result.valid) {
      return new Response(JSON.stringify({
        error: 'Unauthorized',
        message: result.reason,
      }), {
        status: 401,
        headers: { 'Content-Type': 'application/json' },
      });
    }

    // Attach user info to the request for downstream handlers
    const authenticatedRequest = new Request(request);
    authenticatedRequest.headers.set('x-user-id', result.payload.sub);
    authenticatedRequest.headers.set('x-user-role', result.payload.role || 'user');

    return handler(authenticatedRequest, env);
  };
}

// Usage:
const protectedRoute = withAuth(async (request, env) => {
  const userId = request.headers.get('x-user-id');
  return new Response(JSON.stringify({
    message: `Hello, user ${userId}! This is a protected route.`,
  }), { status: 200, headers: { 'Content-Type': 'application/json' } });
});

Important: The JWT verification above is simplified. In production, use a proper library like @std/crypto for HMAC verification, or verify the signature against a JWK (JSON Web Key). Never trust a JWT without verifying its cryptographic signature.

3. Security Headers

Security headers tell browsers how to protect your users. Even a simple static site benefits enormously from these headers. Let's add them to every response:

// security-headers.js
function addSecurityHeaders(response) {
  const headers = new Headers(response.headers);

  // Prevent clicking attacks — only allow our own domain in iframes
  headers.set('X-Frame-Options', 'DENY');

  // Prevent MIME-type sniffing
  headers.set('X-Content-Type-Options', 'nosniff');

  // Content Security Policy — only allow scripts/styles from our domain
  headers.set(
    'Content-Security-Policy',
    "default-src 'self'; script-src 'self' 'unsafe-inline'; style-src 'self' 'unsafe-inline' https://fonts.googleapis.com; font-src https://fonts.gstatic.com; img-src 'self' data: https:; connect-src 'self'; frame-ancestors 'none'; base-uri 'self'; form-action 'self'"
  );

  // Referrer policy — only send origin, not full URL
  headers.set('Referrer-Policy', 'strict-origin-when-cross-origin');

  // Enable HTTPS-only cookies
  headers.set('Strict-Transport-Security', 'max-age=31536000; includeSubDomains; preload');

  // Permissions Policy — restrict browser features
  headers.set(
    'Permissions-Policy',
    'camera=(), microphone=(), geolocation=(), payment=()'
  );

  return new Response(response.body, {
    status: response.status,
    statusText: response.statusText,
    headers,
  });
}

Header breakdown:

  • X-Frame-Options: DENY — blocks your site from being embedded in iframes (prevents clickjacking)
  • X-Content-Type-Options: nosniff — prevents browsers from guessing the content type (prevents MIME confusion attacks)
  • Content-Security-Policy — the most powerful header. Defines exactly which resources can load from where. Start restrictive and relax as needed.
  • Referrer-Policy: strict-origin-when-cross-origin — sends only the origin (not full path) to external sites, protecting user privacy
  • Strict-Transport-Security — forces HTTPS for a year. Cloudflare automatically handles this at the edge, but the header reinforces it
  • Permissions-Policy — disables camera, microphone, geolocation, and payment APIs unless you explicitly need them

Apply these headers by wrapping your response: return addSecurityHeaders(yourResponse);

4. CORS (Cross-Origin Resource Sharing)

When your frontend runs on a different domain (or subdomain) than your Workers API, the browser blocks the request by default. CORS headers tell the browser it's safe to proceed.

Here's a CORS middleware that handles both preflight (OPTIONS) and actual requests:

// cors.js
const ALLOWED_ORIGINS = [
  'https://app.example.com',
  'https://www.example.com',
  'http://localhost:3000', // For local development
];

function handleCors(request) {
  const origin = request.headers.get('Origin') || '';
  const isAllowed = ALLOWED_ORIGINS.includes(origin);

  if (request.method === 'OPTIONS') {
    // Preflight request — just return the CORS headers
    return new Response(null, {
      status: 204,
      headers: {
        'Access-Control-Allow-Origin': isAllowed ? origin : '',
        'Access-Control-Allow-Methods': 'GET, POST, PUT, DELETE, OPTIONS',
        'Access-Control-Allow-Headers': 'Content-Type, Authorization, X-Request-ID',
        'Access-Control-Max-Age': '86400', // Cache preflight for 24 hours
      },
    });
  }
}

Common CORS mistakes to avoid:

  • Never use * for Access-Control-Allow-Origin when your API requires credentials (cookies, auth headers). The spec forbids it.
  • Validate the Origin header against an allowlist, not a regex. A regex like .*\.example\.com$ can be bypassed with domains like evil-example.com.
  • Don't echo back any Origin without validation. An attacker can send their own Origin header and get your API to respond with their domain.

5. Input Validation

Every request contains user-controlled data. Always validate it before processing. Here's a practical input validation helper:

// validate.js
function validateJsonBody(request, schema) {
  const contentType = request.headers.get('Content-Type') || '';

  if (!contentType.includes('application/json')) {
    return {
      valid: false,
      error: 'Content-Type must be application/json',
    };
  }

  let body;
  try {
    body = await request.json();
  } catch {
    return { valid: false, error: 'Invalid JSON body' };
  }

  // Validate required fields
  for (const [field, rules] of Object.entries(schema)) {
    if (rules.required && !(field in body)) {
      return { valid: false, error: `Missing required field: ${field}` };
    }

    if (field in body) {
      const value = body[field];

      // Type checking
      if (rules.type) {
        if (rules.type === 'string' && typeof value !== 'string') {
          return { valid: false, error: `Field '${field}' must be a string` };
        }
        if (rules.type === 'number' && typeof value !== 'number') {
          return { valid: false, error: `Field '${field}' must be a number` };
        }
        if (rules.type === 'email' && typeof value === 'string') {
          if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value)) {
            return { valid: false, error: `Field '${field}' must be a valid email` };
          }
        }
      }

      // Length checks
      if (rules.maxLength && value.length > rules.maxLength) {
        return { valid: false, error: `Field '${field}' exceeds max length of ${rules.maxLength}` };
      }

      if (rules.minLength && value.length < rules.minLength) {
        return { valid: false, error: `Field '${field}' must be at least ${rules.minLength} characters` };
      }
    }
  }

  return { valid: true, data: body };
}

// Usage:
const schema = {
  email: { required: true, type: 'email', maxLength: 254 },
  password: { required: true, type: 'string', minLength: 8, maxLength: 128 },
  name: { type: 'string', maxLength: 100 },
};

const result = await validateJsonBody(request, schema);
if (!result.valid) {
  return new Response(JSON.stringify({ error: result.error }), {
    status: 400,
    headers: { 'Content-Type': 'application/json' },
  });
}

Putting It All Together: A Secure Worker Stack

Here's how to combine all five patterns into a single, secure Worker:

// worker.js — Complete secure Worker
addEventListener('fetch', event => {
  event.respondWith(handleRequest(event.request, event.env));
});

async function handleRequest(request, env) {
  // Step 1: CORS check (handles OPTIONS immediately)
  const corsResponse = handleCors(request);
  if (corsResponse) return corsResponse;

  // Step 2: Rate limiting
  const clientIp = request.headers.get('cf-connecting-ip') || 'unknown';
  const limit = await limiter.check(request, clientIp);
  if (!limit.allowed) {
    return new Response(JSON.stringify({ error: 'Rate limit exceeded' }), {
      status: 429,
      headers: { 'Content-Type': 'application/json', 'Retry-After': '60' },
    });
  }

  // Step 3: Route to handler
  const url = new URL(request.url);
  let response;

  if (url.pathname === '/api/data') {
    // Public endpoint — no auth needed
    response = await publicDataHandler(request, env);
  } else if (url.pathname.startsWith('/api/private')) {
    // Protected endpoint — requires auth
    response = await withAuth(privateHandler)(request, env);
  } else {
    response = new Response('Not Found', { status: 404 });
  }

  // Step 4: Add security headers to every response
  const secureResponse = addSecurityHeaders(response);

  // Step 5: Add rate limit headers
  secureResponse.headers.set('X-RateLimit-Remaining', String(limit.remaining));
  secureResponse.headers.set('X-RateLimit-Reset', String(limit.reset));

  return secureResponse;
}

Summary: The Security Checklist

Every Cloudflare Worker should implement at least these controls:

  1. Rate limiting — prevent abuse with KV-based counters
  2. Authentication — verify JWTs before accessing protected routes
  3. Security headers — CSP, HSTS, X-Frame-Options, and more
  4. CORS — origin validation with an explicit allowlist
  5. Input validation — schema-based checking for every request body

These five patterns cover 90% of common web security threats. Implement them in every new Worker, and you'll start with a solid security foundation instead of retrofitting protections later.
