Python File Operations: Read, Write, Create & Manage Files — Complete Beginner's Guide

July 9, 2026 10 min read

Every Python program needs to read and write data at some point. Whether you're building a web scraper that saves results, a CLI tool that reads config files, or an app that logs errors to disk — understanding file operations is fundamental. This guide covers everything from basic open() to production-grade safe file handling.

1. Basic File Reading and Writing

Python's built-in open() function is your gateway to file operations. It takes a filename and a mode, then returns a file object you can read from or write to.

Reading a File — All at Once

# Read entire file into a string
with open("data.txt", "r") as f:
    content = f.read()
    print(content)

The with statement is critical — it automatically closes the file when the block ends, even if an error occurs. Always use with for file operations.

Reading Line by Line

# Read one line at a time (memory efficient for large files)
with open("large_file.log", "r") as f:
    for line in f:
        print(line.strip())  # strip() removes trailing newline

Iterating over a file object yields one line at a time. This is far more memory-efficient than read() for files larger than available RAM.

Getting All Lines as a List

# Read all lines into a list of strings
with open("data.txt", "r") as f:
    lines = f.readlines()
    print(lines[0])     # First line
    print(len(lines))   # Total number of lines

Writing to a File

# Write (overwrites existing content)
with open("output.txt", "w") as f:
    f.write("Hello, Python!\n")
    f.write("Second line\n")

# Append (adds to end of file)
with open("output.txt", "a") as f:
    f.write("This line is appended\n")

Mode "w" truncates the file first (erases everything). Mode "a" opens for appending without clearing existing content.

3. Directory Operations with os

Python's os module handles directory creation, listing, and management:

import os

# Check if a path exists
if os.path.exists("data"):
    print("Directory exists")

# Check specific types
os.path.isfile("config.json")   # Is it a file?
os.path.isdir("data")           # Is it a directory?

# Create a single directory
os.mkdir("logs")

# Create nested directories (like mkdir -p)
os.makedirs("data/exports/csv", exist_ok=True)

# List contents of a directory
files = os.listdir(".")
for name in files:
    print(name)

# Get the current working directory
print(os.getcwd())

# Join paths safely (handles OS-specific separators)
filepath = os.path.join("data", "exports", "output.csv")
print(filepath)  # data/exports/output.csv (Unix)

4. Modern Path Handling with pathlib

Since Python 3.4, pathlib offers an object-oriented approach that's cleaner than string concatenation:

from pathlib import Path

# Create path objects
config = Path("config/settings.json")
home = Path.home()
print(home)  # /home/username

# Chain paths with /
data_dir = Path("data") / "exports" / "2026"
print(data_dir)  # data/exports/2026

# Check existence
if config.exists():
    content = config.read_text()  # One-liner read!
    
# Write in one line
(Path("output.txt")).write_text("Hello!\n")

# Get parts
print(config.stem)   # settings
print(config.suffix) # .json
print(config.parent) # config

# Find all Python files recursively
py_files = list(Path(".").glob("**/*.py"))
for f in py_files:
    print(f.relative_to("."))

5. Binary File Operations

Images, PDFs, and executables need binary mode — text mode adds unwanted encoding/decoding:

# Read and write binary data
with open("image.png", "rb") as f:
    data = f.read()

with open("copy.png", "wb") as f:
    f.write(data)

# Or use pathlib's binary methods
from pathlib import Path
Path("copy.png").write_bytes(Path("image.png").read_bytes())

# Read with specific encoding
with open("notes.txt", "r", encoding="utf-8") as f:
    text = f.read()

6. Production-Safe Patterns

Real-world file handling needs error handling and atomicity:

from pathlib import Path
import tempfile
import shutil

def safe_write(filepath, content):
    """Atomically write a file — never leave a partial write."""
    path = Path(filepath)
    
    # Ensure parent directory exists
    path.parent.mkdir(parents=True, exist_ok=True)
    
    # Write to a temp file first, then rename (atomic on Unix)
    dir_path = path.parent
    with tempfile.NamedTemporaryFile(
        "w", dir=dir_path, delete=False, suffix=".tmp"
    ) as tmp:
        tmp.write(content)
        tmp_path = tmp.name
    
    # Atomic replace
    Path(tmp_path).rename(path)

# Usage
safe_write("config/settings.json", '{"debug": false}')

# Safe read with default fallback
def read_config(path, default=None):
    try:
        return Path(path).read_text()
    except FileNotFoundError:
        print(f"Config not found: {path}, using defaults")
        return default
    except PermissionError:
        print(f"No permission to read: {path}")
        return default

7. Custom Context Managers for Files

Create reusable file-handling patterns with @contextmanager:

from contextlib import contextmanager
from pathlib import Path

@contextmanager
def write_locked(path, mode="w"):
    """Write to a file with automatic newline handling."""
    file = open(path, mode)
    try:
        yield file
    finally:
        file.flush()  # Ensure data is written to disk
        file.close()

# Usage
with write_locked("app.log", "a") as log:
    log.write(f"Event occurred at {time.time()}\n")

8. Practical Project: Simple Log Analyzer

Let's put it all together — a script that reads a log file, counts error types, and writes a summary report:

from pathlib import Path
from collections import Counter
from datetime import datetime

def analyze_logs(log_path, report_path):
    """Read a log file and generate an error summary."""
    log_file = Path(log_path)
    
    if not log_file.exists():
        print(f"Log file not found: {log_path}")
        return
    
    errors = Counter()
    total_lines = 0
    errors_found = 0
    
    # Read line by line for memory efficiency
    with log_file.open("r", encoding="utf-8") as f:
        for line in f:
            total_lines += 1
            line = line.strip()
            
            if "ERROR" in line:
                # Extract error type
                parts = line.split(":", 1)
                if len(parts) > 1:
                    errors[parts[0].strip()] += 1
                    errors_found += 1
            elif "WARNING" in line:
                errors["WARNING"] += 1
    
    # Build report
    report = []
    report.append("=== Log Analysis Report ===")
    report.append(f"Generated: {datetime.now().isoformat()}")
    report.append(f"Total lines: {total_lines}")
    report.append(f"Errors found: {errors_found}")
    report.append("")
    
    if errors:
        report.append("Error breakdown:")
        for error_type, count in errors.most_common(10):
            report.append(f"  {error_type}: {count}")
    
    # Write report atomically
    report_text = "\n".join(report) + "\n"
    Path(report_path).parent.mkdir(parents=True, exist_ok=True)
    Path(report_path).write_text(report_text)
    print(f"Report written to {report_path}")

# Usage
analyze_logs("var/log/app.log", "reports/error_summary.txt")

Summary: Choosing the Right Mode

  • Read text: open("file.txt", "r") or Path.read_text()
  • Overwrite: open("file.txt", "w") or Path.write_text()
  • Append: open("file.txt", "a")
  • Binary: add b — "rb", "wb"
  • Safe writes: use temp file + rename pattern
  • Large files: iterate line by line, never .read()
  • Always use with: guarantees cleanup on errors

File operations are one of the most common tasks in any Python project. Master these patterns and you'll handle everything from simple config files to production log analysis with confidence. The with statement, pathlib, and proper error handling are the three pillars of safe file operations in Python.
