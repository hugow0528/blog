Python Collections Module: defaultdict, deque, Counter & namedtuple

July 19, 2026 10 min read

Python's built-in dict, list, set, and tuple types cover most use cases. But the collections module ships with specialized containers that solve common patterns in one line instead of five. After reading this guide, you'll never write if key not in dict: dict[key] = [] again.

We'll cover the six most useful classes: defaultdict, Counter, deque, namedtuple, OrderedDict, and ChainMap. Every example is production-ready code you can use today.

defaultdict: Never Check if a Key Exists

The #1 time-saver in the collections module. A defaultdict automatically creates a default value when you access a missing key:

from collections import defaultdict

# Instead of this (regular dict):
groups = {}
for name, grade in [("Alice", "A"), ("Bob", "B"), ("Charlie", "A")]:
    if grade not in groups:
        groups[grade] = []
    groups[grade].append(name)

# Do this (defaultdict):
groups = defaultdict(list)
for name, grade in [("Alice", "A"), ("Bob", "B"), ("Charlie", "A")]:
    groups[grade].append(name)
# defaultdict(list, {'A': ['Alice', 'Charlie'], 'B': ['Bob']})

How it works: you pass a callable (not a value!) as the default factory:

# defaultdict with different default types
counts = defaultdict(int)          # Missing keys → 0
counts["apple"] += 1               # counts["apple"] is now 1

mapping = defaultdict(dict)        # Missing keys → {}
mapping["user1"]["name"] = "Hugo"

sets = defaultdict(set)            # Missing keys → set()
sets["fruits"].add("apple")

# Custom default: return a string
message = defaultdict(lambda: "N/A")
message["name"] = "Hugo"
print(message["email"])  # "N/A"

Real-world example — log aggregation:

from collections import defaultdict
from datetime import datetime

logs = [
    ("2026-07-19 10:00", "ERROR", "Database timeout"),
    ("2026-07-19 10:01", "INFO", "User logged in"),
    ("2026-07-19 10:02", "ERROR", "Connection refused"),
    ("2026-07-19 10:03", "WARN", "Slow query"),
    ("2026-07-19 10:04", "ERROR", "Null pointer"),
]

# Group logs by level
by_level = defaultdict(list)
for timestamp, level, message in logs:
    by_level[level].append(f"{timestamp}: {message}")

for level, entries in by_level.items():
    print(f"\n=== {level} ({len(entries)} entries) ===")
    for entry in entries:
        print(f"  {entry}")

Counter: Count Everything

Counter is a dict subclass that tracks how many times each element appears. It's essentially a multiset:

from collections import Counter

# Count elements in any iterable
words = "the cat sat on the mat the cat ran"
word_counts = Counter(words.split())
print(word_counts)
# Counter({'the': 3, 'cat': 2, 'sat': 1, 'on': 1, 'mat': 1, 'ran': 1})

# Most common N items
print(word_counts.most_common(3))
# [('the', 3), ('cat', 2), ('sat', 1)]

# Count characters
char_counts = Counter("mississippi")
print(char_counts)
# Counter({'i': 4, 's': 4, 'p': 2, 'm': 1})

Math operations between Counters are incredibly useful:

# Subtract counters — find what's missing
shopping_list = Counter(["bread", "milk", "eggs", "butter", "cheese"])
pantry = Counter(["bread", "bread", "eggs", "milk"])
need_to_buy = shopping_list - pantry
print(need_to_buy)  # Counter({'butter': 1, 'cheese': 1})

# Add counters — combine inventories
warehouse_a = Counter({"widget": 50, "gadget": 30})
warehouse_b = Counter({"widget": 20, "gadget": 40, "tool": 10})
total = warehouse_a + warehouse_b
print(total)  # Counter({'widget': 70, 'gadget': 70, 'tool': 10})

# Update in-place
colors = Counter(["red", "blue", "red", "green"])
colors.update(["blue", "blue"])  # Counter({'red': 2, 'blue': 3, 'green': 1})

Real-world example — analyze API response codes:

from collections import Counter
import random

# Simulate 1000 API calls
status_codes = [random.choices(
    [200, 201, 301, 400, 403, 404, 500, 502, 503],
    weights=[60, 10, 2, 5, 3, 10, 5, 3, 2]
)[0] for _ in range(1000)]

counts = Counter(status_codes)
print("Status code distribution:")
for code, count in counts.most_common():
    pct = count / len(status_codes) * 100
    bar = "█" * int(pct / 2)
    print(f"  {code}: {count:4d} ({pct:5.1f}%) {bar}")

# Calculate error rate
errors = sum(c for code, c in counts.items() if code >= 400)
print(f"\nError rate: {errors/len(status_codes)*100:.1f}%")

deque: Fast Appends and Pops from Both Ends

A deque (double-ended queue) supports O(1) appends and pops from both ends. Regular lists are O(n) for inserting/removing at the front:

from collections import deque

q = deque(["a", "b", "c"])

# Append and pop from both ends
q.append("d")           # Right end: deque(['a', 'b', 'c', 'd'])
q.appendleft("z")       # Left end: deque(['z', 'a', 'b', 'c', 'd'])
q.pop()                 # Remove right: deque(['z', 'a', 'b', 'c'])
q.popleft()             # Remove left: deque(['a', 'b', 'c'])

# Max length — auto-discards oldest items when full
window = deque(maxlen=3)
window.append(1)  # deque([1])
window.append(2)  # deque([1, 2])
window.append(3)  # deque([1, 2, 3])
window.append(4)  # deque([2, 3, 4]) ← 1 was automatically removed!

# Rotate
d = deque([1, 2, 3, 4, 5])
d.rotate(1)   # [5, 1, 2, 3, 4]  — shift right
d.rotate(-1)  # [1, 2, 3, 4, 5]  — shift back left

Real-world example — sliding window average:

from collections import deque

def moving_average(values, window_size):
    """Calculate moving average with a fixed window."""
    window = deque(maxlen=window_size)
    results = []

    for value in values:
        window.append(value)
        results.append(sum(window) / len(window))

    return results

# Example: smooth noisy sensor readings
readings = [20, 21, 19, 25, 22, 20, 18, 23, 21, 22]
smoothed = moving_average(readings, window_size=3)

for raw, smooth in zip(readings, smoothed):
    print(f"Raw: {raw:3d} → Smoothed: {smooth:.1f}")
# Raw:  20 → Smoothed: 20.0
# Raw:  21 → Smoothed: 20.5
# Raw:  19 → Smoothed: 20.0
# Raw:  25 → Smoothed: 21.7
# ...

Real-world example — BFS graph traversal:

from collections import deque, defaultdict

def bfs(graph, start):
    """Breadth-first search using deque."""
    visited = set()
    queue = deque([start])
    visited.add(start)
    order = []

    while queue:
        node = queue.popleft()
        order.append(node)

        for neighbor in graph.get(node, []):
            if neighbor not in visited:
                visited.add(neighbor)
                queue.append(neighbor)

    return order

# Social network: who knows whom?
network = {
    "Alice": ["Bob", "Charlie"],
    "Bob": ["Alice", "Diana"],
    "Charlie": ["Alice", "Diana", "Eve"],
    "Diana": ["Bob", "Charlie", "Eve"],
    "Eve": ["Charlie", "Diana"],
}

print(bfs(network, "Alice"))
# ['Alice', 'Bob', 'Charlie', 'Diana', 'Eve']

namedtuple: Tuples with Named Fields

A namedtuple is an immutable record type. It's lighter than a class and faster than a dict for fixed-structure data:

from collections import namedtuple

# Define a "class" in one line
Point = namedtuple("Point", ["x", "y"])
p = Point(3, 4)

print(p.x)       # 3 — access by name
print(p[0])      # 3 — still indexable
print(p)         # Point(x=3, y=4) — nice repr

# Unpack like a regular tuple
x, y = p

# Convert to dict
print(p._asdict())  # {'x': 3, 'y': 4}

# Override with _replace (returns new tuple — it's immutable)
p2 = p._replace(x=10)  # Point(x=10, y=4)

Real-world example — parse CSV-like data:

from collections import namedtuple

User = namedtuple("User", ["id", "name", "email", "role"])

# Simulate database rows
raw_rows = [
    (1, "Hugo", "hugo@example.com", "admin"),
    (2, "Alice", "alice@example.com", "user"),
    (3, "Bob", "bob@example.com", "user"),
    (4, "Charlie", "charlie@example.com", "moderator"),
]

users = [User(*row) for row in raw_rows]

# Query by role
admins = [u for u in users if u.role == "admin"]
print(admins)
# [User(id=1, name='Hugo', email='hugo@example.com', role='admin')]

# With defaults (Python 3.7+)
Contact = namedtuple(
    "Contact", ["name", "phone", "email"],
    defaults=["unknown", "unknown"]
)
c = Contact("Hugo")  # Contact(name='Hugo', phone='unknown', email='unknown')

OrderedDict: Dicts That Remember Insertion Order

Note: As of Python 3.7+, regular dict preserves insertion order. OrderedDict is still useful for its extra methods and explicit semantics:

from collections import OrderedDict

# Last moved to end (LRU cache pattern)
cache = OrderedDict()
cache["first"] = 1
cache["second"] = 2
cache["third"] = 3

# Move to end on access (simulating cache hit)
cache.move_to_end("first")
# Now order: second, third, first

# Pop oldest item (evict least recently used)
evicted = cache.popitem(last=False)
print(evicted)  # ('second', 2)

# Compare dicts ignoring order
od1 = OrderedDict([("a", 1), ("b", 2)])
od2 = OrderedDict([("b", 2), ("a", 1)])
print(od1 == od2)  # False — order matters for OrderedDict!

Real-world example — simple LRU cache:

from collections import OrderedDict

class LRUCache:
    """Least Recently Used cache with OrderedDict."""
    def __init__(self, capacity):
        self.cache = OrderedDict()
        self.capacity = capacity

    def get(self, key):
        if key not in self.cache:
            return None
        # Move to end = recently used
        self.cache.move_to_end(key)
        return self.cache[key]

    def put(self, key, value):
        if key in self.cache:
            self.cache.move_to_end(key)
        self.cache[key] = value
        # Evict oldest if over capacity
        if len(self.cache) > self.capacity:
            self.cache.popitem(last=False)

# Usage
cache = LRUCache(3)
cache.put("a", 1)
cache.put("b", 2)
cache.put("c", 3)
cache.get("a")   # Access 'a', making it most recent
cache.put("d", 4)  # Evicts 'b' (least recently used)

print(list(cache.cache.items()))
# [('a', 1), ('c', 3), ('d', 4)]

ChainMap: Merge Multiple Dicts

ChainMap groups multiple dicts together. Lookups check each dict in order — like a scope chain:

from collections import ChainMap

defaults = {"theme": "dark", "lang": "en", "debug": False}
config = {"theme": "light"}
overrides = {"debug": True}

# Chain: overrides → config → defaults
merged = ChainMap(overrides, config, defaults)

print(merged["theme"])  # "light"  (from config)
print(merged["lang"])   # "en"     (from defaults)
print(merged["debug"])  # True     (from overrides)

# Adding a new chain layer (new scope)
runtime = ChainMap({"env": "prod"}, merged)
print(runtime["theme"])  # "light"

Perfect for environment configs:

from collections import ChainMap
import os

# Layered configuration
system_defaults = {
    "db_host": "localhost",
    "db_port": 5432,
    "log_level": "INFO",
    "cache_ttl": 300,
}

env_config = {k: v for k, v in os.environ.items() if k.startswith("APP_")}
# e.g., {"APP_DB_HOST": "db.production.com", "APP_LOG_LEVEL": "WARNING"}

# Normalize env var names
env_config = {k[4:].lower(): v for k, v in env_config.items()}

# Final config: env overrides everything
config = ChainMap(env_config, system_defaults)
print(f"DB: {config['db_host']}:{config['db_port']}")
print(f"Log: {config['log_level']}")

Summary

The collections module ships with Python — no installs needed. These six classes handle patterns you'd otherwise write manually:

  • defaultdict eliminates missing-key checks
  • Counter counts and compares in one line
  • deque gives you O(1) operations on both ends
  • namedtuple creates lightweight named records
  • OrderedDict preserves order with LRU support
  • ChainMap layers multiple dicts into one view

Start by replacing your if key not in dict patterns with defaultdict and your counting loops with Counter. Those two alone will make your code cleaner within a week.
